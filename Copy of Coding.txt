    <script>
        // === PRODUCT DATA ===
        let products = [
            { id: 1, name: "Wireless Earbuds", price: 1499, qty: 50, emoji: "" },
            { id: 2, name: "Smart Watch", price: 2999, qty: 30, emoji: "⌚" },
            { id: 3, name: "Phone Case", price: 399, qty: 100, emoji: "" },
            { id: 4, name: "Bluetooth Speaker", price: 1899, qty: 25, emoji: "" },
            { id: 5, name: "USB-C Cable", price: 299, qty: 200, emoji: "" },
            { id: 6, name: "Power Bank", price: 999, qty: 40, emoji: "" },
        ];

        let cart = {};
        let currentLocation = null;

        // Load saved data
        if(localStorage.getItem('products')) products = JSON.parse(localStorage.getItem('products'));
        
        // === RENDER PRODUCTS ===
        function renderProducts() {
            const grid = document.getElementById('productGrid');
            grid.innerHTML = products.map(p => `
                <div class="product-card">
                    <div class="product-img">${p.emoji}</div>
                    <div class="product-info">
                        <div class="product-name">${p.name}</div>
                        <div class="product-price">₹${p.price}</div>
                        <div class="stock">Stock: ${p.qty} available</div>
                        <button class="add-btn" 
                            onclick="addToCart(${p.id})" 
                            ${p.qty <= 0 ? 'disabled' : ''}>
                            ${p.qty <= 0 ? 'Out of Stock' : 'Add to Cart'}
                        </button>
                    </div>
                </div>
            `).join('');
        }

        // === CART FUNCTIONS ===
        function addToCart(id) {
            const product = products.find(p => p.id === id);
            if(product.qty <= 0) return;
            
            if(cart[id]) {
                if(cart[id].qty < product.qty) cart[id].qty++;
            } else {
                cart[id] = { ...product, qty: 1 };
            }
            updateCartBar();
        }

        function updateCartBar() {
            const items = Object.values(cart);
            const totalItems = items.reduce((sum, item) => sum + item.qty, 0);
            const totalPrice = items.reduce((sum, item) => sum + (item.price * item.qty), 0);
            
            const bar = document.getElementById('cartBar');
            if(totalItems > 0) {
                bar.style.display = 'flex';
                document.getElementById('cartSummary').textContent = `${totalItems} items • ₹${totalPrice}`;
            } else {
                bar.style.display = 'none';
            }
        }

        // === CHECKOUT ===
        function openCheckout() {
            const items = Object.values(cart);
            const total = items.reduce((sum, item) => sum + (item.price * item.qty), 0);
            
            document.getElementById('cartItems').innerHTML = items.map(i => 
                `<div style="display:flex;justify-content:space-between;padding:5px 0;border-bottom:1px solid #f3f4f6;">
                    <span>${i.emoji} ${i.name} × ${i.qty}</span>
                    <span>₹${i.price * i.qty}</span>
                </div>`
            ).join('') + `<div style="display:flex;justify-content:space-between;padding:10px 0;font-weight:700;font-size:1.1rem;">
                <span>Total</span><span>₹${total}</span>
            </div>`;
            
            document.getElementById('checkoutModal').classList.add('active');
        }

        function closeCheckout() {
            document.getElementById('checkoutModal').classList.remove('active');
        }

        // === GPS LOCATION ===
        function getLocation() {
            const status = document.getElementById('locationStatus');
            status.textContent = " Fetching location...";
            status.style.color = "#3b82f6";
            
            if(!navigator.geolocation) {
                status.textContent = "❌ Geolocation not supported";
                status.style.color = "#ef4444";
                return;
            }
            
            navigator.geolocation.getCurrentPosition(
                (pos) => {
                    currentLocation = { lat: pos.coords.latitude, lng: pos.coords.longitude };
                    document.getElementById('custLat').value = currentLocation.lat;
                    document.getElementById('custLng').value = currentLocation.lng;
                    status.textContent = `✅ Location captured: ${currentLocation.lat.toFixed(4)}, ${currentLocation.lng.toFixed(4)}`;
                    status.style.color = "#10b981";
                },
                (err) => {
                    status.textContent = "❌ Location denied. Please enable GPS.";
                    status.style.color = "#ef4444";
                },
                { enableHighAccuracy: true, timeout: 10000 }
            );
        }

        // === PLACE ORDER ===
        function placeOrder() {
            const name = document.getElementById('custName').value.trim();
            const phone = document.getElementById('custPhone').value.trim();
            const address = document.getElementById('custAddress').value.trim();
            
            if(!name || !phone || !address) { alert("Please fill all fields!"); return; }
            if(!currentLocation) { alert("Please get your location first!"); return; }
            
            const items = Object.values(cart);
            const total = items.reduce((sum, item) => sum + (item.price * item.qty), 0);
            
            // Reduce stock
            items.forEach(item => {
                const product = products.find(p => p.id === item.id);
                if(product) product.qty -= item.qty;
            });
            localStorage.setItem('products', JSON.stringify(products));
            
            // Save order
            const orders = JSON.parse(localStorage.getItem('orders') || '[]');
            orders.unshift({
                id: Date.now(),
                date: new Date().toLocaleString(),
                items: items.map(i => ({ name: i.name, qty: i.qty, price: i.price })),
                total,
                customer: { name, phone, address },
                location: currentLocation,
                status: 'pending'
            });
            localStorage.setItem('orders', JSON.stringify(orders));
            
            // Reset
            cart = {};
            currentLocation = null;
            document.getElementById('custName').value = '';
            document.getElementById('custPhone').value = '';
            document.getElementById('custAddress').value = '';
            document.getElementById('locationStatus').textContent = '';
            
            closeCheckout();
            updateCartBar();
            renderProducts();
            showPage('orders');
            alert("✅ Order placed successfully!");
        }

        // === ORDERS PAGE ===
        function renderOrders() {
            const orders = JSON.parse(localStorage.getItem('orders') || '[]');
            const list = document.getElementById('ordersList');
            
            if(orders.length === 0) {
                list.innerHTML = '<p style="color:#6b7280;text-align:center;margin-top:30px;">No orders yet</p>';
                return;
            }
            
            list.innerHTML = orders.map(o => `
                <div class="order-card">
                    <div style="display:flex;justify-content:space-between;margin-bottom:8px;">
                        <span style="font-weight:700;">Order #${String(o.id).slice(-6)}</span>
                        <span class="order-status status-${o.status}">${o.status.toUpperCase()}</span>
                    </div>
                    <div style="font-size:0.85rem;color:#6b7280;margin-bottom:8px;">${o.date}</div>
                    <div style="font-size:0.9rem;margin-bottom:5px;">
                        ${o.items.map(i => `${i.name} ×${i.qty}`).join(', ')}
                    </div>
                    <div style="font-size:0.85rem;color:#6b7280;"> ${o.customer.address}</div>
                    <div style="font-size:0.8rem;color:#3b82f6;margin-top:5px;">
                        ️ Lat: ${o.location.lat.toFixed(4)}, Lng: ${o.location.lng.toFixed(4)}
                    </div>
                    <div style="font-weight:700;margin-top:8px;font-size:1rem;">Total: ₹${o.total}</div>
                </div>
            `).join('');
        }

        // === NAVIGATION ===
        function showPage(page) {
            document.querySelectorAll('.nav-tab').forEach(t => t.classList.remove('active'));
            event?.target?.classList?.add('active');
            
            document.getElementById('shop-page').style.display = page === 'shop' ? 'block' : 'none';
            document.getElementById('orders-page').className = page === 'orders' ? 'orders-section active' : 'orders-section';
            
            if(page === 'orders') renderOrders();
        }

        // Register Service Worker for PWA
        if('serviceWorker' in navigator) {
            navigator.serviceWorker.register('sw.js').catch(() => {});
        }

        // Init
        renderProducts();
    </script>
